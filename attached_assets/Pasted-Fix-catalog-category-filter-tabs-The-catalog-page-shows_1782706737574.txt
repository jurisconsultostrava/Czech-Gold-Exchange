Also fix the backend route in artifacts/api-server/src/routes/catalog.ts to filter by subcat when provided:
typescriptrouter.get('/products', async (req, res) => {
  const { category, subcat, page = '1', limit = '24' } = req.query;
  
  const conditions = [eq(products.active, true)];
  if (category) conditions.push(eq(products.category, category as string));
  if (subcat) conditions.push(eq(products.subcat, subcat as string));
  
  const items = await db.select()
    .from(products)
    .where(and(...conditions))
    .orderBy(products.sortOrder, products.name)
    .limit(Number(limit))
    .offset((Number(page) - 1) * Number(limit));
    
  res.json(items);
});
